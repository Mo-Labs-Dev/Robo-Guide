cmake_minimum_required(VERSION 3.16)
project(robotest_waveshare)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(CURL REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(waveshare_driver_node src/waveshare_driver_node.cpp)
ament_target_dependencies(waveshare_driver_node rclcpp std_msgs sensor_msgs)
target_link_libraries(waveshare_driver_node CURL::libcurl nlohmann_json::nlohmann_json)
install(TARGETS waveshare_driver_node DESTINATION lib/${PROJECT_NAME})
ament_package()
